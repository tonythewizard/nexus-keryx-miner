cmake_minimum_required(VERSION 3.25)
project(nexus_vulkan_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
if(NOT EXISTS "${NEXUS_LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "An explicitly pinned llama.cpp source tree is required")
endif()
# A separate process and a separate library: never replace the stock CUDA ABI.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${NEXUS_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
find_package(SPIRV-Headers CONFIG REQUIRED)
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
# This large host dispatcher is memory-intensive at -O3; shaders remain unchanged.
target_compile_options(ggml-vulkan PRIVATE -O1 -g0)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(bounded_loader "${CMAKE_CURRENT_BINARY_DIR}/bounded-loader/llama-model-loader.cpp")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bounded-loader")
# Patch only an overlay of the pinned Vulkan dependency, never the stock wrapper.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_bounded_upload.py"
  "${NEXUS_LLAMA_SOURCE}/src/llama-model-loader.cpp" "${bounded_loader}" --replace-output
  COMMAND_ERROR_IS_FATAL ANY)
set(lazy_model "${CMAKE_CURRENT_BINARY_DIR}/bounded-loader/llama-model.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_lazy_mmap.py"
  "${NEXUS_LLAMA_SOURCE}/src/llama-model.cpp" "${lazy_model}"
  COMMAND_ERROR_IS_FATAL ANY)
get_target_property(llama_sources llama SOURCES)
get_target_property(llama_source_dir llama SOURCE_DIR)
set(bounded_sources)
set(bounded_loader_count 0)
set(lazy_model_count 0)
foreach(source IN LISTS llama_sources)
  if(source MATCHES "(^|/)llama-model-loader\\.cpp$")
    list(APPEND bounded_sources "${bounded_loader}")
    math(EXPR bounded_loader_count "${bounded_loader_count}+1")
  elseif(source MATCHES "(^|/)llama-model\\.cpp$")
    list(APPEND bounded_sources "${lazy_model}")
    math(EXPR lazy_model_count "${lazy_model_count}+1")
  elseif(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
    list(APPEND bounded_sources "${source}")
  else()
    list(APPEND bounded_sources "${llama_source_dir}/${source}")
  endif()
endforeach()
if(NOT bounded_loader_count EQUAL 1 OR NOT lazy_model_count EQUAL 1)
  message(FATAL_ERROR "Pinned llama target must contain exactly one model loader and model")
endif()
set_property(TARGET llama PROPERTY SOURCES "${bounded_sources}")
target_include_directories(llama PRIVATE "${NEXUS_LLAMA_SOURCE}/src")
find_program(NEXUS_CLANGXX clang++ REQUIRED)
set_property(TARGET ggml-vulkan PROPERTY CXX_COMPILER_LAUNCHER
  "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/clang_dispatcher.py")
add_library(nexus-llama-vulkan SHARED runtime.cpp)
target_include_directories(nexus-llama-vulkan PRIVATE ../../include)
target_link_libraries(nexus-llama-vulkan PRIVATE llama ggml-vulkan)
target_link_options(nexus-llama-vulkan PRIVATE -Wl,--no-keep-memory -Wl,--no-undefined)
set_target_properties(nexus-llama-vulkan PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(nexus-vulkan-inference-probe probe.cpp)
target_compile_features(nexus-vulkan-inference-probe PRIVATE cxx_std_20)
target_include_directories(nexus-vulkan-inference-probe PRIVATE ../../include)
target_link_libraries(nexus-vulkan-inference-probe PRIVATE OpenSSL::Crypto ${CMAKE_DL_LIBS})
